# Reglas de negocio y operación

Las reglas siguientes fijan decisiones observables. Los valores detallados de recursos, alarmas y despliegue se precisarán en sus etapas correspondientes.

```yaml source-of-truth
rules:
  - id: "BR1.1"
    statement: "Ruta y acceso"
    category: "authorization"
    applies_to: "Solicitud HTTP"
    trigger: "Al recibir POST"
    logic: "IF la ruta es /telemetry-events y la clave API es válida THEN continuar; ELSE rechazar sin encolar."
    violation_behaviour: "Rechazar sin publicar."
    source: "FR1.1, FR2.1"
  - id: "BR1.2"
    statement: "Campos e identificadores"
    category: "validation"
    applies_to: "TelemetryEvent"
    trigger: "Antes de encolar"
    logic: "IF faltan campos obligatorios o eventId/vehicleId están vacíos o tienen tipo no textual THEN rechazar; no exigir UUID."
    violation_behaviour: "HTTP 400 sin publicar."
    source: "FR1.2"
  - id: "BR1.3"
    statement: "Tipos y rangos"
    category: "validation"
    applies_to: "TelemetryEvent"
    trigger: "Antes de encolar"
    logic: "IF eventType no es battery, temperature o speed, value no es finito, o está fuera de su rango [0,100], [-50,150] o [0,300] respectivamente THEN rechazar."
    violation_behaviour: "HTTP 400 sin publicar."
    source: "FR1.3, FR1.4"
  - id: "BR1.4"
    statement: "Fecha del evento"
    category: "validation"
    applies_to: "TelemetryEvent"
    trigger: "Antes de encolar"
    logic: "IF timestamp no es ISO 8601 UTC o supera el reloj del servicio en más de cinco minutos THEN rechazar; exactamente cinco minutos se acepta."
    violation_behaviour: "HTTP 400 sin publicar."
    source: "FR1.5"
  - id: "BR1.5"
    statement: "Aceptación correlacionada"
    category: "policy"
    applies_to: "Solicitud HTTP"
    trigger: "Tras publicación confirmada en SQS de entrada"
    logic: "IF SQS confirma THEN responder 202 con status=accepted, msg_id=correlationId y acceptedAt UTC; transportar el mismo acceptedAt y correlationId en el mensaje."
    violation_behaviour: "Nunca responder 202 antes de confirmar."
    source: "FR1.6, FR3.1"
  - id: "BR1.6"
    statement: "Fallo de encolado"
    category: "policy"
    applies_to: "Solicitud HTTP"
    trigger: "Cuando SQS falla o su resultado es incierto"
    logic: "IF no hay confirmación de publicación THEN responder error no exitoso y registrar correlationId."
    violation_behaviour: "No afirmar aceptación."
    source: "FR1.6"
  - id: "BR1.7"
    statement: "Límite por clave"
    category: "constraint"
    applies_to: "Clave API"
    trigger: "En cada solicitud"
    logic: "IF una clave supera 100 solicitudes/s sostenidas, ráfaga 200 o cuota diaria 5000000 THEN limitar."
    violation_behaviour: "Respuesta de límite; no publicar."
    source: "FR2.2"
  - id: "BR2.1"
    statement: "Persistencia idempotente"
    category: "constraint"
    applies_to: "TelemetryEvent"
    trigger: "Al consumir SQS"
    logic: "IF eventId no existe THEN escribir condicionalmente; IF existe con payload canónico idéntico THEN tratar como duplicado sin sobrescribir."
    violation_behaviour: "Conservar el primer evento."
    source: "FR4.1, FR4.2"
  - id: "BR2.2"
    statement: "Conflicto de idempotencia"
    category: "constraint"
    applies_to: "TelemetryEvent"
    trigger: "Al encontrar eventId existente"
    logic: "IF el contenido canónico difiere THEN registrar IDEMPOTENCY_CONFLICT con correlación y métrica; no modificar evento ni salida."
    violation_behaviour: "Error del mensaje sin mutación."
    source: "FR4.2, FR4.4"
  - id: "BR2.3"
    statement: "Payload y retención"
    category: "policy"
    applies_to: "TelemetryEvent"
    trigger: "Al persistir"
    logic: "IF el evento fue aceptado THEN guardar rawPayload sin modificar, acceptedAt original y expiresAt=acceptedAt+90 días."
    violation_behaviour: "Reintentar si faltan datos canónicos."
    source: "FR4.1, NFR4"
  - id: "BR2.4"
    statement: "Salida atómica"
    category: "constraint"
    applies_to: "TelemetryEvent y AlertOutbox"
    trigger: "Al procesar batería baja"
    logic: "IF battery.value<20 THEN crear evento y salida PENDING con alertId determinista en una sola transacción; ELSE crear solo evento."
    violation_behaviour: "Ningún estado parcial."
    source: "FR4.3, FR5.1"
  - id: "BR2.5"
    statement: "Duplicado pendiente"
    category: "constraint"
    applies_to: "AlertOutbox"
    trigger: "Al reentregar un evento idéntico"
    logic: "IF la salida ya existe THEN conservarla y no cambiar su estado ni su alertId."
    violation_behaviour: "No crear otra salida."
    source: "FR4.4"
  - id: "BR3.1"
    statement: "Umbral estricto"
    category: "calculation"
    applies_to: "TelemetryEvent"
    trigger: "Al evaluar alertas"
    logic: "IF eventType=battery AND value<20 THEN requiere alerta; value=20, otros tipos y eventos fuera de orden no alteran alertas anteriores."
    violation_behaviour: "Sin alerta para los demás casos."
    source: "FR5.1"
  - id: "BR3.2"
    statement: "Contenido de alerta"
    category: "validation"
    applies_to: "BatteryAlert"
    trigger: "Al construir publicación"
    logic: "IF hay alerta THEN incluir alertId, eventId, vehicleId, alertType=LOW_BATTERY, batteryValue, occurredAt UTC y correlationId; mantener alertId estable."
    violation_behaviour: "No publicar mensaje incompleto."
    source: "FR5.2, FR5.3"
  - id: "BR3.3"
    statement: "Publicación recuperable"
    category: "policy"
    applies_to: "AlertOutbox"
    trigger: "Mientras status=PENDING"
    logic: "IF la publicación falla o es incierta THEN conservar PENDING y reintentar con el mismo alertId; admitir reentregas físicas."
    violation_behaviour: "No perder la alerta lógica."
    source: "FR5.3, FR5.4"
  - id: "BR3.4"
    statement: "Confirmación condicional"
    category: "constraint"
    applies_to: "AlertOutbox"
    trigger: "Tras confirmación de SQS"
    logic: "IF SQS confirma THEN cambiar condicionalmente PENDING a PUBLISHED; un duplicado no revierte ni adelanta el estado."
    violation_behaviour: "Conservar PENDING ante incertidumbre."
    source: "FR5.4"
  - id: "BR4.1"
    statement: "Fallo parcial y DLQ"
    category: "policy"
    applies_to: "Lote de mensajes"
    trigger: "Al procesar SQS"
    logic: "IF falla un mensaje THEN informar solo ese fallo; tras cinco recepciones fallidas moverlo a DLQ sin bloquear los sanos."
    violation_behaviour: "Reintento acotado."
    source: "FR6.1, FR6.2"
  - id: "BR4.2"
    statement: "Registro de fallos"
    category: "policy"
    applies_to: "Procesamiento"
    trigger: "Al fallar"
    logic: "IF hay error THEN registrar correlationId, eventId disponible, recepción y causa sin secretos."
    violation_behaviour: "No registrar claves API."
    source: "FR6.3"
  - id: "BR4.3"
    statement: "Redrive seguro"
    category: "authorization"
    applies_to: "DLQ"
    trigger: "Antes y durante redrive"
    logic: "IF el actor y alcance están autorizados THEN mostrar selección, cantidad, velocidad y destino, exigir confirmación y registrar resultado por lote; la idempotencia evita efectos duplicados."
    violation_behaviour: "Rechazar alcance vacío o actor no autorizado."
    source: "FR4.2, FR6.2"
  - id: "BR5.1"
    statement: "Objetivos de servicio"
    category: "policy"
    applies_to: "Servicio"
    trigger: "En medición operativa"
    logic: "IF se mide staging THEN evaluar p95<=500ms a 100 solicitudes/s, 99% persistidos en menos de 30s y disponibilidad mensual >=99.9%."
    violation_behaviour: "Emitir evidencia y alarma de incumplimiento."
    source: "NFR1, NFR2, NFR3"
  - id: "BR5.2"
    statement: "Señales y alarmas"
    category: "policy"
    applies_to: "Servicio"
    trigger: "En operación"
    logic: "IF ocurren errores, pendientes, atraso, DLQ o TTL anómalo THEN emitir métricas/logs correlacionados y alarmas con umbral, ventana, severidad, destino y recuperación."
    violation_behaviour: "Investigar por correlationId."
    source: "NFR7"
  - id: "BR6.1"
    statement: "Infraestructura y seguridad"
    category: "policy"
    applies_to: "Recursos AWS"
    trigger: "Al desplegar"
    logic: "IF se aprovisiona staging THEN definir recursos en CDK, cifrar SQS/DynamoDB/CloudTrail con KMS, usar TLS e IAM mínimo y conservar auditoría 90 días."
    violation_behaviour: "Bloquear despliegue no conforme."
    source: "NFR5, NFR9, NFR10"
  - id: "BR6.2"
    statement: "Entrega controlada"
    category: "policy"
    applies_to: "Pipeline"
    trigger: "Antes de integrar o promover"
    logic: "IF CI no supera formato, lint, tipos, pruebas, dependencias y secretos THEN bloquear integración; usar OIDC, squash a main, staging automático y producción con aprobación manual."
    violation_behaviour: "No promover cambios sin controles."
    source: "NFR6, NFR8"
```
